  if(NOT H3_VENDOR_DEPENDENCIES)

  endif()
  add_library(h3_audio native/src/audio.cpp)
  target_include_directories(h3_audio PUBLIC native/include)
  target_link_libraries(h3_audio PUBLIC h3_core SDL2::SDL2)
  target_compile_options(h3_audio PRIVATE -Wall -Wextra -Wpedantic)
  if(BUILD_TESTING)
    add_executable(h3_audio_tests native/tests/audio_tests.cpp)
    target_link_libraries(h3_audio_tests PRIVATE h3_audio)
    target_compile_definitions(h3_audio_tests PRIVATE H3_TEST_SOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME native_audio COMMAND h3_audio_tests)
    set_tests_properties(native_audio PROPERTIES ENVIRONMENT "SDL_AUDIODRIVER=dummy")
  endif()

    target_link_libraries(HeroesIII PRIVATE SDL2::SDL2main)
    set_target_properties(HeroesIII PROPERTIES
      MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/native/ios/Info.plist.in"
      XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "2"
      XCODE_ATTRIBUTE_ASSETCATALOG_COMPILER_APPICON_NAME AppIcon
      XCODE_ATTRIBUTE_CODE_SIGN_STYLE Automatic
      XCODE_ATTRIBUTE_SUPPORTED_PLATFORMS "iphoneos iphonesimulator"
      XCODE_ATTRIBUTE_ENABLE_BITCODE NO)

      set_target_properties(HeroesIII PROPERTIES XCODE_ATTRIBUTE_DEVELOPMENT_TEAM "${H3_DEVELOPMENT_TEAM}")

    target_sources(HeroesIII PRIVATE native/macos/native_pointer.mm)
    target_sources(HeroesIII PRIVATE native/macos/Assets/HeroesIII.icns)
    set_source_files_properties(native/macos/Assets/HeroesIII.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    target_sources(HeroesIII PRIVATE native/assets/app-icon.png)
    set_source_files_properties(native/assets/app-icon.png PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    set_source_files_properties(native/macos/native_pointer.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
    set_target_properties(HeroesIII PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED ON)
    target_link_libraries(HeroesIII PRIVATE "-framework Cocoa")
    # Seal Info.plist and the icon with the executable after every link. The
    # linker's bare ad-hoc signature does not describe a complete app bundle.
    add_custom_command(TARGET HeroesIII POST_BUILD
      COMMAND /usr/bin/codesign --force --sign - "$<TARGET_BUNDLE_DIR:HeroesIII>"
      VERBATIM)
  else()
    target_sources(HeroesIII PRIVATE native/src/native_pointer.cpp)
  endif()

    target_compile_definitions(HeroesIII PRIVATE H3_SOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
  else()
    target_compile_definitions(HeroesIII PRIVATE H3_SOURCE_ROOT="")
  endif()
  target_compile_options(HeroesIII PRIVATE -Wall -Wextra -Wpedantic)
  set_target_properties(HeroesIII PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "local.heroesiii.native" MACOSX_BUNDLE_BUNDLE_NAME "Heroes III" MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}")
  if(NOT H3_IOS)
    set_target_properties(HeroesIII PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/native/macos/Info.plist.in")
  endif()
  if(H3_RUNTIME_DATA_DIR)
    if(H3_IOS)
      # A native resource phase follows Xcode's build/archive destinations and
      # makes resource changes invalidate the bundle's code signature.
      if(CMAKE_VERSION VERSION_LESS 4.1)
        message(FATAL_ERROR "iPad GameData folder resources require CMake 4.1 or newer")
      endif()
      target_sources(HeroesIII PRIVATE "${H3_RUNTIME_DATA_DIR}")
      set_source_files_properties("${H3_RUNTIME_DATA_DIR}" PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    else()
      set(h3_bundle_data "$<TARGET_BUNDLE_DIR:HeroesIII>/Contents/Resources/GameData")
      add_custom_command(TARGET HeroesIII PRE_LINK
        COMMAND "${CMAKE_COMMAND}" -E copy_directory "${H3_RUNTIME_DATA_DIR}" "${h3_bundle_data}"
        VERBATIM)
    endif()
  endif()
endif()
